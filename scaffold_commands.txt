# ==============================================================================
# COMANDOS DE EF CORE SCAFFOLD (DATABASE FIRST)
# ==============================================================================
# Observação: Certifique-se de que a ferramenta dotnet-ef esteja instalada:
# dotnet tool install --global dotnet-ef (ou dotnet tool update --global dotnet-ef)
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. AUTH SERVICE
# ------------------------------------------------------------------------------
# Connection String: Host=localhost;Port=5433;Database=auth_db;Username=auth_user;Password=auth_pass
# DbContext: AuthDbContext
# Provedor: Npgsql.EntityFrameworkCore.PostgreSQL

# Execute dentro da pasta /AuthService:
dotnet ef dbcontext scaffold "Host=localhost;Port=5433;Database=auth_db;Username=auth_user;Password=auth_pass" Npgsql.EntityFrameworkCore.PostgreSQL --context AuthDbContext --output-dir Infrastructure/Persistence/Entities --context-dir Infrastructure/Persistence/Context --force

# Ou execute da raiz da solução apontando para o projeto:
dotnet ef dbcontext scaffold "Host=localhost;Port=5433;Database=auth_db;Username=auth_user;Password=auth_pass" Npgsql.EntityFrameworkCore.PostgreSQL --project ./AuthService/AuthService.csproj --context AuthDbContext --output-dir Infrastructure/Persistence/Entities --context-dir Infrastructure/Persistence/Context --force
